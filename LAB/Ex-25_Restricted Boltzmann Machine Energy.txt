import numpy as np
from sklearn.neural_network import BernoulliRBM

X = np.random.randint(0,2,(100,6))

rbm = BernoulliRBM(n_components=3, learning_rate=0.1, n_iter=10, random_state=42)
rbm.fit(X)

hidden = rbm.transform(X)
reconstructed = np.dot(hidden, rbm.components_) + rbm.intercept_visible_

print("Hidden Representation:\n", hidden[:5])
print("Reconstructed Data:\n", reconstructed[:5])
